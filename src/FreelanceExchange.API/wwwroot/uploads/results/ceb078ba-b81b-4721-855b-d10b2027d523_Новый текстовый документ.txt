    <script>
        lucide.createIcons();

        const token = localStorage.getItem('token');
        const user = JSON.parse(localStorage.getItem('user') || '{}');
        if (!token) window.location.href = 'login.html';
        if (user.role !== 'Customer') { alert('Эта страница только для заказчиков'); window.location.href = 'index.html'; }

        document.getElementById('userNameHeading').innerText = user.fullName || 'Заказчик';
        document.getElementById('logoutBtn').addEventListener('click', () => { localStorage.clear(); window.location.href = 'index.html'; });

        // Вкладки
        let currentTab = 'myOrders';
        const tabs = document.querySelectorAll('.tab-btn');
        const contents = {
            myOrders: document.getElementById('myOrdersTab'),
            responses: document.getElementById('responsesTab'),
            finance: document.getElementById('financeTab')
        };
        function switchTab(tabId) {
            Object.values(contents).forEach(c => c.classList.add('hidden'));
            contents[tabId].classList.remove('hidden');
            tabs.forEach(btn => {
                if (btn.dataset.tab === tabId) btn.classList.add('tab-active');
                else btn.classList.remove('tab-active');
            });
            currentTab = tabId;
            if (tabId === 'myOrders') loadMyOrders();
            if (tabId === 'responses') loadAllResponses();
            if (tabId === 'finance') { loadTransactions(); loadBalance(); }
        }
        tabs.forEach(btn => btn.addEventListener('click', () => switchTab(btn.dataset.tab)));
        switchTab('myOrders');

        async function loadBalance() {
            try {
                const res = await fetch('/api/Finances', { headers: { 'Authorization': `Bearer ${token}` } });
                if (res.ok) { const data = await res.json(); document.getElementById('balance').innerText = `${data.balance.toFixed(2)} BYN`; }
            } catch(e) {}
        }

        // Мои заказы
        async function loadMyOrders() {
            const container = document.getElementById('myOrdersTab');
            container.innerHTML = '<div class="text-center py-6 text-sm text-приглушенный">Загрузка...</div>';
            try {
                const res = await fetch('/api/Orders', { headers: { 'Authorization': `Bearer ${token}` } });
                if (res.ok) {
                    let orders = await res.json();
                    orders = orders.items || orders;
                    if (!orders.length) { container.innerHTML = '<div class="премиум-карта p-8 text-center">У вас пока нет заказов</div>'; return; }
                    container.innerHTML = orders.map(order => `
                        <div class="премиум-карта rounded-xl p-5">
                            <div class="flex justify-between items-start flex-wrap gap-3">
                                <div><h3 class="font-bold">${escapeHtml(order.title)}</h3><p class="text-приглушенный text-sm">${escapeHtml(order.description)}</p>
                                <div class="flex gap-3 mt-2 text-xs"><span>💰 ${order.budget} BYN</span><span>📅 ${new Date(order.deadline).toLocaleDateString()}</span><span>Статус: ${order.status}</span></div></div>
                                <div class="flex gap-2">
                                    ${order.status === 'Open' ? `<button class="view-responses-btn bg-винныйАкцент/20 text-винныйАкцент px-3 py-1 rounded text-xs" data-id="${order.id}">Отклики</button>
                                                               <button class="delete-order-btn bg-red-500/20 text-red-400 px-3 py-1 rounded text-xs" data-id="${order.id}">Удалить</button>` : ''}
                                    ${order.status === 'InProgress' ? `<button class="complete-order-btn bg-green-500/20 text-green-400 px-3 py-1 rounded text-xs" data-id="${order.id}">Завершить</button>` : ''}
                                </div>
                            </div>
                        </div>
                    `).join('');
                    document.querySelectorAll('.view-responses-btn').forEach(btn => btn.addEventListener('click', () => { switchTab('responses'); }));
                    document.querySelectorAll('.delete-order-btn').forEach(btn => btn.addEventListener('click', async () => { if(confirm('Удалить?')){ await fetch(`/api/Orders/${btn.dataset.id}`, { method: 'DELETE', headers: { 'Authorization': `Bearer ${token}` } }); loadMyOrders(); } }));
                    document.querySelectorAll('.complete-order-btn').forEach(btn => btn.addEventListener('click', async () => { if(confirm('Завершить заказ? Средства перейдут фрилансеру')){ await fetch(`/api/Orders/${btn.dataset.id}/complete`, { method: 'POST', headers: { 'Authorization': `Bearer ${token}` } }); loadMyOrders(); } }));
                } else throw new Error();
            } catch(e) { container.innerHTML = '<div class="text-center text-red-400">Ошибка загрузки</div>'; }
        }

        // Отклики на все заказы
        async function loadAllResponses() {
            const container = document.getElementById('responsesTab');
            container.innerHTML = '<div class="text-center py-6">Загрузка...</div>';
            try {
                const ordersRes = await fetch('/api/Orders', { headers: { 'Authorization': `Bearer ${token}` } });
                let orders = await ordersRes.json();
                orders = orders.items || orders;
                let allResponses = [];
                for (let order of orders) {
                    const respRes = await fetch(`/api/Responses/order/${order.id}`, { headers: { 'Authorization': `Bearer ${token}` } });
                    if (respRes.ok) {
                        let responses = await respRes.json();
                        responses.forEach(r => r.orderTitle = order.title);
                        allResponses.push(...responses);
                    }
                }
                if (!allResponses.length) { container.innerHTML = '<div class="премиум-карта p-8 text-center">Нет откликов на ваши заказы</div>'; return; }
                container.innerHTML = allResponses.map(r => `
                    <div class="премиум-карта rounded-xl p-5">
                        <div><strong>Заказ:</strong> ${escapeHtml(r.orderTitle)}</div>
                        <div><strong>Фрилансер:</strong> ${escapeHtml(r.freelancerName)}</div>
                        <div><strong>Предложено:</strong> ${r.proposedPrice} BYN</div>
                        <div class="mt-2">${r.coverLetter}</div>
                        <div class="mt-3 flex gap-2">
                            ${r.status === 'Pending' ? `<button class="accept-response bg-green-500/20 text-green-400 px-3 py-1 rounded text-xs" data-response="${r.id}" data-order="${r.orderId}">Принять</button>` : ''}
                            <span class="text-xs">Статус: ${getStatus(r.status)}</span>
                        </div>
                    </div>
                `).join('');
                document.querySelectorAll('.accept-response').forEach(btn => btn.addEventListener('click', async () => {
                    const respId = btn.dataset.response;
                    await fetch(`/api/Responses/${respId}/accept`, { method: 'POST', headers: { 'Authorization': `Bearer ${token}` } });
                    alert('Отклик принят, заказ передан фрилансеру');
                    loadAllResponses(); loadMyOrders();
                }));
            } catch(e) { container.innerHTML = '<div class="text-red-400">Ошибка</div>'; }
        }

        async function loadTransactions() {
            const container = document.getElementById('transactionsList');
            try {
                const res = await fetch('/api/Finances/transactions', { headers: { 'Authorization': `Bearer ${token}` } });
                if (res.ok) {
                    let tx = await res.json();
                    if (!tx.length) { container.innerHTML = '<div class="text-приглушенный text-sm py-4">Нет операций</div>'; return; }
                    container.innerHTML = tx.map(t => `<div class="flex justify-between py-3"><span>${t.type}</span><span class="font-mono">${t.amount} BYN</span><span class="text-xs text-приглушенный">${new Date(t.date).toLocaleDateString()}</span></div>`).join('');
                } else container.innerHTML = '<div class="text-приглушенный">Ошибка загрузки</div>';
            } catch(e) { container.innerHTML = '<div class="text-приглушенный">Ошибка сети</div>'; }
        }

        // Пополнение
        document.getElementById('depositBtn').addEventListener('click', () => document.getElementById('depositModal').classList.remove('hidden'));
        document.getElementById('closeDepositModal').addEventListener('click', () => document.getElementById('depositModal').classList.add('hidden'));
        document.getElementById('depositConfirmBtn').addEventListener('click', async () => {
            const amount = parseFloat(document.getElementById('depositAmount').value);
            if (isNaN(amount) || amount <= 0) { alert('Введите сумму'); return; }
            try {
                const res = await fetch('/api/Finances/deposit', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` }, body: JSON.stringify({ amount }) });
                if (res.ok) { alert('Баланс пополнен'); loadBalance(); document.getElementById('depositModal').classList.add('hidden'); }
                else alert('Ошибка');
            } catch(e) { alert('Ошибка сети'); }
        });

        // Создание заказа
        let categoriesLoaded = false;
        async function loadCategories() {
            const select = document.getElementById('orderCategory');
            select.innerHTML = '<option value="">Загрузка...</option>';
            try {
                const res = await fetch('/api/Admin/public/categories');
                if (res.ok) {
                    const cats = await res.json();
                    select.innerHTML = '';
                    cats.forEach(cat => {
                        select.innerHTML += `<option value="${cat.id}">${cat.name}</option>`;
                    });
                    categoriesLoaded = true;
                } else {
                    select.innerHTML = '<option value="">Ошибка загрузки категорий</option>';
                }
            } catch(e) {
                select.innerHTML = '<option value="">Ошибка сети</option>';
            }
        }
        document.getElementById('createOrderBtn').addEventListener('click', () => {
            loadCategories();

        document.getElementById('createOrderForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const title = document.getElementById('orderTitle').value;
            const description = document.getElementById('orderDesc').value;
            const budget = parseFloat(document.getElementById('orderBudget').value);
            const deadline = new Date(document.getElementById('orderDeadline').value).toISOString();
            const categoryId = parseInt(document.getElementById('orderCategory').value);
            const errorDiv = document.getElementById('orderError');
            errorDiv.classList.add('hidden');
            if (!title || !description || isNaN(budget) || !deadline || isNaN(categoryId)) {
                errorDiv.textContent = 'Заполните все поля';
                errorDiv.classList.remove('hidden');
                return;
            }
            try {
                const res = await fetch('/api/Orders', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
                    body: JSON.stringify({ title, description, budget, deadline, categoryId })
                });
                if (res.ok) {
                    alert('Заказ создан!');
                    document.getElementById('createOrderModal').classList.add('hidden');
                    loadMyOrders();
                } else {
                    const err = await res.text();
                    errorDiv.textContent = err || 'Ошибка создания';
                    errorDiv.classList.remove('hidden');
                }
            } catch(e) {
                errorDiv.textContent = 'Ошибка сети';
                errorDiv.classList.remove('hidden');
            }
        });

        document.getElementById('changeEmailBtn').addEventListener('click', async () => {
            const newEmail = document.getElementById('newEmail').value.trim();
            const password = document.getElementById('emailPassword').value;
            if (!newEmail || !password) { alert('Заполните поля'); return; }
            const res = await fetch('/api/Users/email', { method: 'PUT', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` }, body: JSON.stringify({ newEmail, password }) });
            if (res.ok) { alert('Email изменён, перезайдите'); localStorage.clear(); window.location.href = 'login.html'; }
            else alert('Ошибка');
        });
        document.getElementById('changePasswordBtn').addEventListener('click', async () => {
            const oldPass = document.getElementById('oldPassword').value;
            const newPass = document.getElementById('newPassword').value;
            const confirm = document.getElementById('confirmPassword').value;
            if (!oldPass || !newPass || newPass !== confirm) { alert('Пароли не совпадают'); return; }
            const res = await fetch('/api/Users/change-password', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` }, body: JSON.stringify({ oldPassword: oldPass, newPassword: newPass }) });
            if (res.ok) { alert('Пароль изменён, перезайдите'); localStorage.clear(); window.location.href = 'login.html'; }
            else alert('Ошибка');
        });
